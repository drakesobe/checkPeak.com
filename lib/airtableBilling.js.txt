// lib/airtableBilling.js
import Airtable from "airtable";

const BILLING_API_KEY = process.env.BILLING_API_KEY;
const BILLING_BASE_ID = process.env.BILLING_BASE_ID;
const BILLING_TABLE = process.env.BILLING_TABLE_NAME; // can be table id

if (!BILLING_API_KEY || !BILLING_BASE_ID || !BILLING_TABLE) {
  // don't throw on import in serverless; just warn
  console.warn("[billing] Missing billing env vars.");
}

const base = new Airtable({ apiKey: BILLING_API_KEY }).base(BILLING_BASE_ID);

export const Billing = () => base(BILLING_TABLE);

/**
 * Column names (adjust ONLY if your Airtable names differ)
 */
export const F = {
  Organization: "Organization", // link to Organizations (single)
  StripeCustomerId: "Stripe Customer ID",
  StripeSubscriptionId: "Stripe Subscription ID",
  Plan: "Plan",
  BillingStatus: "Billing Status",
  CurrentPeriodEnd: "Current Period End",
  TrialEnds: "Trial Ends",
  PaymentMethodType: "Default Payment Method Type",
  PaymentMethodLast4: "Payment Method Last 4",
  PaymentMethodBrand: "Payment Method Brand",
  Currency: "Currency",
  LastSynced: "Last Synced From Stripe",
  BillingContactName: "Billing Contact Name",
  BillingEmail: "Billing Email",
  BillingPhone: "Billing Phone",
  LegalBusinessName: "Legal Business Name",
  PaymentTerms: "Payment Terms",
  InvoiceNotes: "Invoice Notes",
  BillingNotes: "Billing Notes",
};

export async function findBillingRecordByOrgId(orgRecordId) {
  const formula = `FIND("${orgRecordId}", ARRAYJOIN({${F.Organization}}&""))>0`;
  const res = await Billing()
    .select({ maxRecords: 1, filterByFormula: formula })
    .firstPage();
  return res?.[0] || null;
}

export async function findBillingRecordByStripeCustomerId(customerId) {
  const formula = `{${F.StripeCustomerId}}="${customerId}"`;
  const res = await Billing()
    .select({ maxRecords: 1, filterByFormula: formula })
    .firstPage();
  return res?.[0] || null;
}

export async function upsertBillingForOrg(orgRecordId, fields) {
  const existing = await findBillingRecordByOrgId(orgRecordId);
  const patch = {
    ...fields,
    [F.Organization]: [orgRecordId],
    [F.LastSynced]: new Date().toISOString(),
  };

  if (existing) {
    const updated = await Billing().update(existing.id, patch);
    return updated;
  }

  const created = await Billing().create(patch);
  return created;
}

export function isoDateFromUnixSeconds(sec) {
  if (!sec) return "";
  const d = new Date(sec * 1000);
  // Airtable Date field likes ISO; date-only fields also accept ISO
  return d.toISOString();
}
